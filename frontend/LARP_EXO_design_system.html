Overview

LARP EXO is a retro arcade visual language. Every decision flows from three constraints: gold on black, zero border-radius, and Press Start 2P everywhere.

⚔ GOLD BORDER
STANDARD BORDER
COMMON UNCOMMON RARE EPIC 👑 LEGENDARY
⭐ JACKPOT METER
✓ DO
  • Gold on black, always
  • Zero border-radius everywhere
  • Press Start 2P for all UI chrome
  • Space Mono for readable body text
  • Solid pixel shadows, no blur
✗ DON'T
  • Add rounded corners anywhere
  • Use any font other than these two
  • Use blur-based shadows
  • Invent new rarity tiers
  • Use rarity colors for non-rarity UI

Spacing & Radius

Spacing is built on an 8px grid (base unit = 4px, always use multiples of 2). Radius is always zero.

Border radius — always 0

radius.base = "0rem"
Hard corners are fundamental. No border-radius anywhere in the system.
This applies to containers, buttons, inputs, badges, and all custom components.

PixelBorder.tsx

Drop-in component. Requires Tailwind + CSS custom properties from tokens.

import * as React from "react"

// If you're NOT using the design system's cn() util, replace with:
// function cn(...classes) { return classes.filter(Boolean).join(' ') }
import { cn } from "./utils"

export interface PixelBorderProps extends React.HTMLAttributes<HTMLDivElement> {
  /** gold — full gold border + 4px pixel drop-shadow. */
  gold?: boolean
}

export function PixelBorder({
  className, gold = false, children, style, ...props
}: PixelBorderProps) {
  return (
    <div
      className={cn(
        "border-2 bg-[#0d0d0d]",
        gold ? "border-[#ffd700]" : "border-[#ffd700]/35",
        className
      )}
      style={{
        boxShadow: gold
          ? "4px 4px 0 #b8860b"
          : "2px 2px 0 #b8860b44",
        borderRadius: 0,  // always zero
        ...style,
      }}
      {...props}
    >
      {children}
    </div>
  )
}

PixelRarityBadge.tsx

Self-contained — no Tailwind dependency for rarity colors (all inline styles).

import * as React from "react"

export type Rarity = "common" | "uncommon" | "rare" | "epic" | "legendary"

const RARITY_CONFIG: Record<Rarity, {
  border: string; bg: string; text: string; glow: string; label: string
}> = {
  common:    { border:"#666",    bg:"#111",    text:"#888",    glow:"none",                label:"COMMON"    },
  uncommon:  { border:"#4a9eff", bg:"#05111f", text:"#4a9eff", glow:"0 0 8px #4a9eff44", label:"UNCOMMON"  },
  rare:      { border:"#a855f7", bg:"#0e0518", text:"#a855f7", glow:"0 0 8px #a855f755", label:"RARE"      },
  epic:      { border:"#ec4899", bg:"#1a0511", text:"#ec4899", glow:"0 0 10px #ec489966",label:"EPIC"      },
  legendary: { border:"#ffd700", bg:"#140d00", text:"#ffd700", glow:"0 0 12px #ffd70077",label:"LEGENDARY" },
}

export function PixelRarityBadge({
  rarity, icon, style, ...props
}: { rarity: Rarity; icon?: string } & React.HTMLAttributes<HTMLSpanElement>) {
  const cfg = RARITY_CONFIG[rarity]
  return (
    <span
      style={{
        display:"inline-flex", alignItems:"center", gap:4,
        border:"1px solid", borderColor:cfg.border, borderRadius:0,
        padding:"3px 8px", background:cfg.bg, color:cfg.text,
        boxShadow:cfg.glow,
        fontFamily:"'Press Start 2P', monospace", fontSize:7,
        letterSpacing:"0.05em", textTransform:"uppercase",
        ...style,
      }}
      {...props}
    >
      {icon && <span style={{fontSize:10}}>{icon}</span>}
      {cfg.label}
    </span>
  )
}
